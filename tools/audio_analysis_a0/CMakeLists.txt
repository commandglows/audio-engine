cmake_minimum_required(VERSION 3.16)
project(shipglows_analysis_a0 LANGUAGES CXX)
include_directories(${CMAKE_CURRENT_LIST_DIR}/../../flutter/shipglows_audio/native/engine/include)
add_executable(audio_analysis_a0 calibration.cpp)
target_compile_features(audio_analysis_a0 PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(audio_analysis_a0 PRIVATE /W4 /fp:precise)
else()
 target_compile_options(audio_analysis_a0 PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
enable_testing()
if(WIN32)
  add_executable(audio_analysis_a0_etl ../audio_analysis_a0_windows/review_heap_etl.cpp)
  target_compile_features(audio_analysis_a0_etl PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(audio_analysis_a0_etl PRIVATE -municode)
  endif()
  target_link_libraries(audio_analysis_a0_etl PRIVATE advapi32 tdh)
 add_test(NAME a0_etl_ledger_controls COMMAND audio_analysis_a0_etl --self-test)
endif()
add_test(NAME a0_reference COMMAND audio_analysis_a0 --test)
add_executable(audio_analysis_a0_allocations allocations.cpp)
target_compile_features(audio_analysis_a0_allocations PRIVATE cxx_std_17)
add_test(NAME a0_cpp_new_allocations COMMAND audio_analysis_a0_allocations)
set(A0_FLUTTER_WRAPPER "" CACHE PATH "Explicit Flutter cpp_client_wrapper for private bridge allocation measurement")
if(A0_FLUTTER_WRAPPER)
 add_executable(audio_analysis_a0_bridge_allocations allocations.cpp "${A0_FLUTTER_WRAPPER}/standard_codec.cc")
 target_compile_features(audio_analysis_a0_bridge_allocations PRIVATE cxx_std_17)
 target_compile_definitions(audio_analysis_a0_bridge_allocations PRIVATE A0_BRIDGE_ALLOCATIONS)
 target_include_directories(audio_analysis_a0_bridge_allocations PRIVATE "${A0_FLUTTER_WRAPPER}/include")
 add_test(NAME a0_bridge_cpp_new_allocations COMMAND audio_analysis_a0_bridge_allocations)
endif()
